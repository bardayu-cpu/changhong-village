## 03 自己写的厨房 v3 接入探针（跑完删，不提交）：两种厅各走一遍——宴会厅 → 后厅 → 厨房（界面按钮、盘位、文案）→ 回宴会厅，钱包、厨房进度、换场景
const Main = preload("res://scripts/main.gd")
const DT := 1.0 / 60.0
const KSAVE := "user://suqi_kitchen.json"
const KBAK := "user://suqi_kitchen.bak03"
const GAMES := "user://tmp03_games.json"

var m
var started := false
var ok := true

func _initialize() -> void:
    if FileAccess.file_exists(KSAVE):
        DirAccess.copy_absolute(ProjectSettings.globalize_path(KSAVE), ProjectSettings.globalize_path(KBAK))
        DirAccess.remove_absolute(ProjectSettings.globalize_path(KSAVE))
    DirAccess.remove_absolute(ProjectSettings.globalize_path(Main.HALL55_FLAG))
    DirAccess.remove_absolute(ProjectSettings.globalize_path(GAMES))
    m = Main.new()
    m.fast = true
    get_root().add_child(m)

func _process(_dt: float) -> bool:
    if started:
        return false
    started = true
    run()
    return false

func check(name: String, cond: bool) -> void:
    print(("  好 " if cond else "  不行 ") + name)
    ok = ok and cond

func tick(n: int) -> void:
    for i in n:
        m.cv._process(DT)
        m._process(DT)

func walk_to(target: Vector2, secs := 30.0) -> void:
    m.cv.go_to(target)
    for i in int(secs / DT):
        tick(1)
        if m.cv.path.is_empty():
            break

func all_text(n: Node, out: Array = []) -> Array:
    if n is Label:
        out.append((n as Label).text)
    elif n is Button:
        out.append((n as Button).text)
    for c in n.get_children():
        all_text(c, out)
    return out

func nodes_with_meta(n: Node, key: String, out: Array = []) -> Array:
    if n.has_meta(key):
        out.append(n)
    for c in n.get_children():
        nodes_with_meta(c, key, out)
    return out

func press(p, action: String) -> bool:
    for b in nodes_with_meta(p.playfield, "kitchen_action"):
        if b is Button and str(b.get_meta("kitchen_action")) == action and not b.disabled:
            b.pressed.emit()
            return true
    return false

func greedy(p, max_steps := 2400) -> void:
    var g = p.rules
    for step in max_steps:
        if not g.active:
            break
        for i in g.pots.size():
            var pot: Dictionary = g.pots[i]
            if pot.state == "empty":
                var taken := {}
                for q in g.pots:
                    if q.state != "empty":
                        taken["%s:%s" % [q.order_id, q.dish_index]] = true
                for pl in g.plates:
                    taken["%s:%s" % [pl.order_id, pl.dish_index]] = true
                var os: Array = g.orders.duplicate()
                os.sort_custom(func(a, b): return float(a.left) < float(b.left))
                var picked := false
                for o in os:
                    if picked:
                        break
                    for d in o.dishes.size():
                        if not o.dishes[d].done and not taken.has("%s:%s" % [o.id, d]) and g.choose_order_dish(i, o.id, d):
                            for ing in g.pots[i].required:
                                g.add_ingredient(i, ing)
                            picked = true
                            break
            elif pot.state == "ready":
                if g.serve_pot(i):
                    g.deliver_auto()
            elif pot.state in ["burnt", "invalid"]:
                g.clear_pot(i)
        while not g.plates.is_empty() and g.deliver_auto():
            pass
        g.tick(0.5)
    p.departures.clear()
    p._process(0.01)

    walk_to(b.spot_pos("to_hall"))

            break

func run() -> void:
    m.cloud.fake_no_news = true
    for i in 600:
        if m.state == "title":
            break
        await process_frame
    await process_frame
    m.games_save.local_path = GAMES
    m.court_save.local_path = "user://tmp03_court.json"
    m.nickname = "小李"
    await m._enter_banquet()
    await m._set_hall55(false, false)   # 默认厅现在是 55 桌，要走第 44 版厅得明着切
    await create_timer(0.3).timeout
    check("先走第 44 版厅（桌数 %d，big=%s）" % [m.hall.cfg["tables"].size(), m.hall.big], not m.hall.big and m.hall.cfg["tables"].size() < 20)
    await route("第 44 版厅")
    await m._set_hall55(true, false)
    await create_timer(0.5).timeout
    check("切到 55 桌厅", m.hall.big and m.state == "banquet")
    await route("55 桌厅")
    pass
    DirAccess.remove_absolute(ProjectSettings.globalize_path(Main.HALL55_FLAG))
    DirAccess.remove_absolute(ProjectSettings.globalize_path(GAMES))
    DirAccess.remove_absolute(ProjectSettings.globalize_path("user://tmp03_court.json"))
    DirAccess.remove_absolute(ProjectSettings.globalize_path(KSAVE))
    if FileAccess.file_exists(KBAK):
        DirAccess.copy_absolute(ProjectSettings.globalize_path(KBAK), ProjectSettings.globalize_path(KSAVE))
        DirAccess.remove_absolute(ProjectSettings.globalize_path(KBAK))
    print("RESULT ", "PASS" if ok else "FAIL")
    quit(0 if ok else 1)
